cmake_minimum_required(VERSION 3.25)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" CACHE STRING "Static Windows runtime")
project(BetterKuGouAudioHost VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BK_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../../.cache/audio-effects/deps" CACHE PATH "Pinned dependency cache")
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_WAYLAND_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_USE_STATIC_CRT ON CACHE BOOL "" FORCE)
add_subdirectory("${BK_DEPS}/vst3sdk" "${CMAKE_CURRENT_BINARY_DIR}/sdk" EXCLUDE_FROM_ALL)
set(SDK "${BK_DEPS}/vst3sdk")
add_library(bk_fft STATIC "${BK_DEPS}/kissfft/kiss_fft.c" "${BK_DEPS}/kissfft/kiss_fftr.c")
target_include_directories(bk_fft PUBLIC "${BK_DEPS}/kissfft")
add_executable(bk-audio-host src/main.cpp src/plugin.cpp src/tempo.cpp
  "${SDK}/public.sdk/source/vst/hosting/plugprovider.cpp"
  "${SDK}/public.sdk/source/common/memorystream.cpp")
target_include_directories(bk-audio-host PRIVATE "${SDK}" "${BK_DEPS}/json/single_include" src)
target_link_libraries(bk-audio-host PRIVATE sdk_hosting bk_fft)
if(WIN32)
  target_sources(bk-audio-host PRIVATE src/platform_win.cpp "${SDK}/public.sdk/source/vst/hosting/module_win32.cpp")
  target_compile_definitions(bk-audio-host PRIVATE NOMINMAX UNICODE _UNICODE)
  target_compile_options(bk-audio-host PRIVATE /utf-8 /W4)
  target_link_libraries(bk-audio-host PRIVATE user32 ole32 shell32)
elseif(APPLE)
  enable_language(OBJCXX)
  target_sources(bk-audio-host PRIVATE src/platform_mac.mm "${SDK}/public.sdk/source/vst/hosting/module_mac.mm")
  set_source_files_properties(src/platform_mac.mm "${SDK}/public.sdk/source/vst/hosting/module_mac.mm" PROPERTIES COMPILE_FLAGS "-fobjc-arc")
  target_link_libraries(bk-audio-host PRIVATE "-framework Cocoa")
else()
  find_package(X11 REQUIRED)
  target_sources(bk-audio-host PRIVATE src/platform_linux.cpp "${SDK}/public.sdk/source/vst/hosting/module_linux.cpp")
  target_include_directories(bk-audio-host PRIVATE ${X11_INCLUDE_DIR})
  target_link_libraries(bk-audio-host PRIVATE ${X11_LIBRARIES} ${CMAKE_DL_LIBS} pthread)
endif()
if(BK_OUTPUT_DIRECTORY)
  set_target_properties(bk-audio-host PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${BK_OUTPUT_DIRECTORY}" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${BK_OUTPUT_DIRECTORY}" RUNTIME_OUTPUT_DIRECTORY_DEBUG "${BK_OUTPUT_DIRECTORY}")
endif()
enable_testing()
add_test(NAME tempo-and-protocol COMMAND bk-audio-host --self-test)
